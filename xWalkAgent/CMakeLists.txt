cmake_minimum_required(VERSION 3.16)

project(xWalkAgent VERSION 1.0.0 LANGUAGES CXX)

option(XWALK_AGENT_BUILD_HOST "Build every xWalk agent module and deterministic host test" OFF)
option(XWALK_AGENT_BUILD_RPI "Build every xWalk agent module and hardware-labelled executable" OFF)

if(XWALK_AGENT_BUILD_HOST AND XWALK_AGENT_BUILD_RPI)
    message(FATAL_ERROR "Host and RPI verification require separate build directories")
endif()

set(XWALK_PICARX_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_PICARX_BUILD_HARDWARE_TESTS ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)
set(XWALK_LINE_TRACKING_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_LINE_TRACKING_BUILD_HARDWARE_TESTS ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)
set(XWALK_SELF_DRIVE_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_SELF_DRIVE_BUILD_HARDWARE_TESTS ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)
set(XWALK_LOCAL_VOICE_CHATBOT_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_CAMERA_CAPTURE_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_CAMERA_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_CAMERA_BUILD_HARDWARE_TESTS ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)
set(XWALK_CAMERA_BUILD_LINUX_BACKEND ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Build Linux camera capture for the RPi Agent" FORCE)
set(XWALK_VOICE_ACTIVE_CAR_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_VOICE_ACTIVE_CAR_GPT_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_BOOT_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_BOOT_BUILD_RPI_BACKEND ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)
set(XWALK_SPI_TRANSFER_BUILD_HOST_TESTS ${XWALK_AGENT_BUILD_HOST} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_HOST" FORCE)
set(XWALK_SPI_BUILD_LINUX_BACKEND ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Build Linux SPI ownership for the RPi Agent" FORCE)
set(XWALK_SPI_BUILD_HARDWARE_TESTS ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Controlled by XWALK_AGENT_BUILD_RPI" FORCE)

    "Build ALSA music composition for the RPi Agent" FORCE)

    "Build native compressed-audio decoding for the RPi Agent" FORCE)

    "Build ALSA speech capture for the RPi Agent" FORCE)

    "Build ALSA speech playback for the RPi Agent" FORCE)

    "Build the Vosk provider for the RPi Agent" FORCE)
set(XWALK_GPT_BUILD_ESPEAK_PROVIDER ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Build the Espeak provider for the RPi Agent" FORCE)
set(XWALK_LANGUAGE_MODEL_BUILD_OLLAMA_PROVIDER ${XWALK_AGENT_BUILD_RPI} CACHE BOOL
    "Build the Ollama provider for the RPi Agent" FORCE)

    XWALK_LINE_TRACKING_BUILD_HARDWARE_TESTS
    XWALK_SELF_DRIVE_BUILD_HOST_TESTS
    XWALK_SELF_DRIVE_BUILD_HARDWARE_TESTS

    XWALK_GPT_BUILD_ESPEAK_PROVIDER
    XWALK_LANGUAGE_MODEL_BUILD_OLLAMA_PROVIDER
    XWALK_CAMERA_BUILD_LINUX_BACKEND

if(XWALK_AGENT_BUILD_HOST OR XWALK_AGENT_BUILD_RPI)
    set(BUILD_TESTING ON CACHE BOOL "Enable xWalk Agent aggregate tests" FORCE)
    include(CTest)
endif()

add_subdirectory(xWalkPicarx)
add_subdirectory(xWalkLineTracking)
add_subdirectory(xWalkSelfDrive)
add_subdirectory(xWalkLocalVoiceChatbot)
add_subdirectory(xWalkCameraCapture)
add_subdirectory(xWalkVoiceActiveCar)
add_subdirectory(xWalkVoiceActiveCarGpt)
add_subdirectory(xWalkSpiTransfer)
add_subdirectory(xWalkBoot)

add_library(xWalkAgent INTERFACE)
add_library(xWalk::Agent ALIAS xWalkAgent)
target_compile_features(xWalkAgent INTERFACE cxx_std_17)
target_link_libraries(xWalkAgent INTERFACE
    xWalkPicarx
    xWalkLineTracking
    xWalkSelfDrive
    xWalkLocalVoiceChatbot
    xWalkCameraCapture
    xWalkVoiceActiveCar
    xWalkVoiceActiveCarGpt
    xWalkSpiTransfer
